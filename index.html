/** * BLEND THEORY REFERRAL SYSTEM * Refer 2 completed clients = 1 free standard haircut */ const SPREADSHEET_ID = '10mPAkNid8xGFZMPnT13d6vXAB6a1i7XcGG1G1qKzkec'; const CLIENTS_SHEET = 'Clients'; const REFERRALS_SHEET = 'Referrals'; const BOOKING_URL = 'https://blend-theory-103153.square.site/'; /* ========================= DISPLAY REFERRAL PAGE ========================= */ function doGet(e) { const referralCode = String( (e && e.parameter && (e.parameter.ref || e.parameter.code)) || '' ).trim().toUpperCase(); const template = HtmlService.createTemplate(getReferralPage_()); template.referralCode = referralCode; return template.evaluate() .setTitle('Blend Theory Referral') .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } /* ========================= RECEIVE GITHUB SUBMISSION ========================= */ function doPost(e) { try { let data = {}; /* * GitHub sends JSON as text/plain. * Parse that JSON here. */ if ( e && e.postData && e.postData.contents ) { try { data = JSON.parse(e.postData.contents); } catch (parseError) { data = e.parameter || {}; } } else { data = e.parameter || {}; } const result = submitReferral(data); return ContentService .createTextOutput( JSON.stringify(result) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { return ContentService .createTextOutput( JSON.stringify({ ok: false, message: error && error.message ? error.message : 'Referral error.' }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ========================= PROCESS REFERRAL ========================= */ function submitReferral(data) { try { data = data || {}; const referralCode = String(data.referralCode || '') .trim() .toUpperCase(); const clientName = String(data.name || '').trim(); const phone = String(data.phone || '').trim(); const email = String(data.email || '').trim(); /* Validate submission */ if (!referralCode) { throw new Error( 'Missing referral code.' ); } if (!clientName) { throw new Error( 'Please enter your name.' ); } if (!phone && !email) { throw new Error( 'Please enter a phone number or email.' ); } /* Open spreadsheet */ const ss = SpreadsheetApp.openById( SPREADSHEET_ID ); const clients = ss.getSheetByName( CLIENTS_SHEET ); const referrals = ss.getSheetByName( REFERRALS_SHEET ); if (!clients || !referrals) { throw new Error( 'Required spreadsheet tab was not found.' ); } /* ========================= FIND REFERRING CLIENT ========================= */ const clientLastRow = Math.max( clients.getLastRow(), 2 ); const clientData = clients .getRange( 2, 1, clientLastRow - 1, 4 ) .getDisplayValues(); let referrerName = ''; for ( let i = 0; i < clientData.length; i++ ) { const code = String( clientData[i][3] || '' ) .trim() .toUpperCase(); if (code === referralCode) { referrerName = clientData[i][1]; break; } } if (!referrerName) { throw new Error( 'Referral code not recognized.' ); } /* ========================= CHECK FOR DUPLICATE ========================= */ const referralLastRow = Math.max( referrals.getLastRow(), 2 ); const referralRows = referrals .getRange( 2, 1, referralLastRow - 1, 11 ) .getDisplayValues(); const normalizedPhone = phone.replace(/\D/g, ''); const normalizedEmail = email.toLowerCase(); for ( let i = 0; i < referralRows.length; i++ ) { const existingCode = String( referralRows[i][1] || '' ) .trim() .toUpperCase(); const existingContact = String( referralRows[i][4] || '' ) .toLowerCase(); if ( existingCode !== referralCode ) { continue; } const samePhone = normalizedPhone && existingContact .replace(/\D/g, '') .includes( normalizedPhone ); const sameEmail = normalizedEmail && existingContact.includes( normalizedEmail ); if ( samePhone || sameEmail ) { return { ok: true, duplicate: true, message: 'This referral has already been recorded.', bookingUrl: buildBookingUrl_( referralCode ) }; } } /* ========================= FIND EMPTY REFERRAL ROW ========================= Columns A, C and J contain formulas, so we determine whether a row is actually unused by checking B and D. */ let targetRow = 0; const maxRows = referrals.getMaxRows(); for ( let row = 2; row <= maxRows; row++ ) { const code = referrals .getRange(row, 2) .getDisplayValue(); const newClient = referrals .getRange(row, 4) .getDisplayValue(); if ( !code && !newClient ) { targetRow = row; break; } } /* * If every existing row is occupied, * add another row. */ if (!targetRow) { referrals.insertRowAfter( maxRows ); targetRow = maxRows + 1; } /* ========================= RECORD REFERRAL ========================= */ const now = new Date(); const contact = [phone, email] .filter(Boolean) .join(' | '); /* * Write only user-entered fields. * * A = formula * B = referral code * C = formula/referrer * D = new client * E = contact * F = date referred * G = booking date * H = status * I = completion date * J = formula/link * K = notes */ referrals .getRange(targetRow, 2) .setValue( referralCode ); referrals .getRange(targetRow, 4) .setValue( clientName ); referrals .getRange(targetRow, 5) .setValue( contact ); referrals .getRange(targetRow, 6) .setValue( now ); referrals .getRange(targetRow, 8) .setValue( 'Invited' ); referrals .getRange(targetRow, 11) .setValue( 'Submitted from Blend Theory referral page' ); SpreadsheetApp.flush(); return { ok: true, duplicate: false, message: 'Referral recorded.', referrerName: referrerName, bookingUrl: buildBookingUrl_( referralCode ) }; } catch (err) { return { ok: false, message: err && err.message ? err.message : 'Unable to save referral.' }; } } /* ========================= SQUARE BOOKING URL ========================= */ function buildBookingUrl_( referralCode ) { return ( BOOKING_URL + '?utm_source=referral' + '&utm_medium=client' + '&utm_campaign=refer2free' + '&utm_content=' + encodeURIComponent( referralCode ) ); } /* ========================= DIRECT APPS SCRIPT PAGE This remains available for troubleshooting, although customers will use GitHub. ========================= */ function getReferralPage_() { return ` Blend Theory Referral
BLEND THEORY
CLIENT REFERRAL
You've been put on.

A Blend Theory client sent you here. Claim the referral and book your cut.

Referral code:
Referral credit is awarded after the referred client completes and pays for a qualifying appointment.
`; }
